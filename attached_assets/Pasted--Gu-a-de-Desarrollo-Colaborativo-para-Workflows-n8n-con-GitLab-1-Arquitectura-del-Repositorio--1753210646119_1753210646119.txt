### 3.2 Script de Sincronización Colaborativa

```javascript
// scripts/sync.js
const axios = require('axios');
const fs = require('fs').promises;
const path = require('path');

class N8nCollaborativeSync {
  constructor(environment) {
    this.env = environment;
    this.config = require(`../config/environments.json`)[environment];
    this.apiKey = process.env[`N8N_${environment.toUpperCase()}_API_KEY`];
    this.baseURL = process.env[`N8N_${environment.toUpperCase()}_URL`];
  }

  async syncWorkflows() {
    console.log(`🔄 Sincronizando workflows para ${this.env}...`);
    
    try {
      // 1. Obtener workflows del entorno
      const workflows = await this.getRemoteWorkflows();
      
      // 2. Sincronizar con archivos locales
      await this.syncToLocal(workflows);
      
      // 3. Generar metadata de sincronización
      await this.generateSyncMetadata();
      
      console.log(`✅ Sincronización completada: ${workflows.length} workflows`);
    } catch (error) {
      console.error(`❌ Error en sincronización: ${error.message}`);
      process.exit(1);
    }
  }

  async getRemoteWorkflows() {
    const response = await axios.get(`${this.baseURL}/api/v1/workflows`, {
      headers: { 'X-N8N-API-KEY': this.apiKey }
    });
    return response.data.data;
  }

  async syncToLocal(workflows) {
    const envDir = path.join('workflows', this.env);
    await fs.mkdir(envDir, { recursive: true });
    
    for (const workflow of workflows) {
      const filename = `${workflow.name.replace(/[^a-zA-Z0-9]/g, '_')}_${workflow.id}.json`;
      const filepath = path.join(envDir, filename);
      
      // Añadir metadata de sincronización
      const workflowWithMeta = {
        ...workflow,
        syncMetadata: {
          syncDate: new Date().toISOString(),
          environment: this.env,
          gitCommit: process.env.CI_COMMIT_SHA || 'local',
          syncedBy: process.env.GITLAB_USER_EMAIL || 'local-user'
        }
      };
      
      await fs.writeFile(filepath, JSON.stringify(workflowWithMeta, null, 2));
      console.log(`💾 Sincronizado: ${filename}`);
    }
  }

  async generateSyncMetadata() {
    const metadata = {
      lastSync: new Date().toISOString(),
      environment: this.env,
      totalWorkflows: (await this.getRemoteWorkflows()).length,
      gitCommit: process.env.CI_COMMIT_SHA || 'local',
      pipelineId: process.env.CI_PIPELINE_ID || null
    };
    
    await fs.writeFile(
      path.join('workflows', this.env, '_sync_metadata.json'),
      JSON.stringify(metadata, null, 2)
    );
  }
}

// Ejecutar sincronización
const env = process.argv.find(arg => arg.startsWith('--env='))?.split('=')[1] || 'development';
new N8nCollaborativeSync(env).syncWorkflows();
```

### 3.3 Script de Deploy Colaborativo

```javascript
// scripts/deploy.js
const axios = require('axios');
const fs = require('fs').promises;
const path = require('path');
const Joi = require('joi');

class CollaborativeDeploy {
  constructor(environment) {
    this.env = environment;
    this.apiKey = process.env[`N8N_${environment.toUpperCase()}_API_KEY`];
    this.baseURL = process.env[`N8N_${environment.toUpperCase()}_URL`];
    this.deployLog = [];
  }

  async deployWorkflows() {
    console.log(`🚀 Iniciando deploy a ${this.env}...`);
    
    try {
      // 1. Validar archivos antes del deploy
      await this.validateWorkflows();
      
      // 2. Obtener workflows a deployar
      const workflows = await this.getWorkflowsToDeploy();
      
      // 3. Deploy incremental
      for (const workflow of workflows) {
        await this.deployWorkflow(workflow);
      }
      
      // 4. Generar reporte de deploy
      await this.generateDeployReport();
      
      console.log(`✅ Deploy completado: ${workflows.length} workflows`);
    } catch (error) {
      console.error(`❌ Deploy fallido: ${error.message}`);
      await this.rollback();
      process.exit(1);
    }
  }

  async deployWorkflow(workflowFile) {
    const workflow = JSON.parse(await fs.readFile(workflowFile, 'utf8'));
    const workflowName = workflow.name;
    
    try {
      // Verificar si existe el workflow
      const existingId = await this.findExistingWorkflow(workflowName);
      
      let deployedWorkflow;
      if (existingId) {
        // Actualizar workflow existente
        deployedWorkflow = await this.updateWorkflow(existingId, workflow);
        console.log(`📝 Actualizado: ${workflowName} (ID: ${existingId})`);
      } else {
        // Crear nuevo workflow
        deployedWorkflow = await this.createWorkflow(workflow);
        console.log(`✨ Creado: ${workflowName} (ID: ${deployedWorkflow.id})`);
      }
      
      // Activar si está marcado como activo
      if (workflow.active) {
        await this.activateWorkflow(deployedWorkflow.id);
        console.log(`🟢 Activado: ${workflowName}`);
      }
      
      // Log del deploy
      this.deployLog.push({
        workflow: workflowName,
        id: deployedWorkflow.id,
        action: existingId ? 'updated' : 'created',
        active: workflow.active,
        timestamp: new Date().toISOString()
      });
      
    } catch (error) {
      console.error(`❌ Error deploying ${workflowName}: ${error.message}`);
      throw error;
    }
  }

  async findExistingWorkflow(name) {
    const response = await axios.get(`${this.baseURL}/api/v1/workflows`, {
      headers: { 'X-N8N-API-KEY': this.apiKey }
    });
    
    const existing = response.data.data.find(w => w.name === name);
    return existing ? existing.id : null;
  }

  async createWorkflow(workflow) {
    const response = await axios.post(`${this.baseURL}/api/v1/workflows`, workflow, {
      headers: { 
        'X-N8N-API-KEY': this.apiKey,
        'Content-Type': 'application/json'
      }
    });
    return response.data;
  }

  async updateWorkflow(id, workflow) {
    const response = await axios.put(`${this.baseURL}/api/v1/workflows/${id}`, workflow, {
      headers: { 
        'X-N8N-API-KEY': this.apiKey,
        'Content-Type': 'application/json'
      }
    });
    return response.data;
  }

  async activateWorkflow(id) {
    await axios.post(`${this.baseURL}/api/v1/workflows/${id}/activate`, {}, {
      headers: { 'X-N8N-API-KEY': this.apiKey }
    });
  }

  async generateDeployReport() {
    const report = {
      environment: this.env,
      deployDate: new Date().toISOString(),
      gitCommit: process.env.CI_COMMIT_SHA,
      pipelineUrl: process.env.CI_PIPELINE_URL,
      deployedBy: process.env.GITLAB_USER_EMAIL,
      workflows: this.deployLog
    };
    
    await fs.writeFile(
      `deploy-report-${this.env}-${Date.now()}.json`,
      JSON.stringify(report, null, 2)
    );
  }
}

// Ejecutar deploy
const env = process.argv.find(arg => arg.startsWith('--env='))?.split('=')[1];
if (!env) {
  console.error('❌ Especifica el entorno: --env=development|staging|production');
  process.exit(1);
}

new CollaborativeDeploy(env).deployWorkflows();
```

### 5.3 Revisión Automatizada

```javascript
// scripts/validate.js - Validación colaborativa
const Joi = require('joi');
const fs = require('fs').promises;
const path = require('path');

const workflowSchema = Joi.object({
  name: Joi.string().pattern(/^[A-Z][a-zA-Z0-9_]*$/).required(),
  nodes: Joi.array().min(1).required(),
  connections: Joi.object().required(),
  active: Joi.boolean().default(false),
  // Metadata obligatorio para colaboración
  metadata: Joi.object({
    author: Joi.string().email().required(),
    version: Joi.string().pattern(/^\d+\.\d+\.\d+$/).required(),
    description: Joi.string().min(10).required(),
    category: Joi.string().valid('etl', 'webhook', 'scheduled', 'manual').required(),
    dependencies: Joi.array().items(Joi.string()),
    tags: Joi.array().items(Joi.string()),
    lastModified: Joi.date().iso().required(),
    reviewedBy: Joi.string().email(),
    approvedBy: Joi.string().email()
  }).required()
});

class WorkflowValidator {
  async validateAll() {
    const errors = [];
    const workflowDirs = ['development', 'staging', 'production'];
    
    for (const dir of workflowDirs) {
      const dirPath = path.join('workflows', dir);
      if (await this.pathExists(dirPath)) {
        const dirErrors = await this.validateDirectory(dirPath);
        errors.push(...dirErrors);
      }
    }
    
    if (errors.length > 0) {
      console.error('❌ Errores de validación encontrados:');
      errors.forEach(error => console.error(`  - ${error}`));
      process.exit(1);
    }
    
    console.log('✅ Todos los workflows son válidos');
  }

  async validateDirectory(dirPath) {
    const errors = [];
    const files = await fs.readdir(dirPath);
    
    for (const file of files) {
      if (file.endsWith('.json') && !file.startsWith('_')) {
        const filePath = path.join(dirPath, file);
        const fileErrors = await this.validateWorkflow(filePath);
        errors.push(...fileErrors);
      }
    }
    
    return errors;
  }

  async validateWorkflow(filePath) {
    const errors = [];
    
    try {
      const content = await fs.readFile(filePath, 'utf8');
      const workflow = JSON.parse(content);
      
      // Validación de esquema
      const { error } = workflowSchema.validate(workflow);
      if (error) {
        errors.push(`${filePath}: ${error.details[0].message}`);
      }
      
      // Validaciones específicas de n8n
      const n8nErrors = this.validateN8nWorkflow(workflow, filePath);
      errors.push(...n8nErrors);
      
    } catch (parseError) {
      errors.push(`${filePath}: JSON inválido - ${parseError.message}`);
    }
    
    return errors;
  }

  validateN8nWorkflow(workflow, filePath) {
    const errors = [];
    
    // Validar que no haya credenciales hardcodeadas
    const workflowStr = JSON.stringify(workflow);
    const suspiciousPatterns = [
      /password['"]\s*:\s*['"][^'"]{8,}/gi,
      /api[_-]?key['"]\s*:\s*['"][^'"]{10,}/gi,
      /secret['"]\s*:\s*['"][^'"]{8,}/gi,
      /token['"]\s*:\s*['"][^'"]{10,}/gi
    ];
    
    suspiciousPatterns.forEach(pattern => {
      if (pattern.test(workflowStr)) {
        errors.push(`${filePath}: Posibles credenciales hardcodeadas detectadas`);
      }
    });
    
    // Validar estructura de nodos
    if (!workflow.nodes || workflow.nodes.length === 0) {
      errors.push(`${filePath}: Workflow sin nodos`);
    }
    
    // Validar nodo de error handling
    const hasErrorHandler = workflow.nodes.some(node => 
      node.type === 'n8n-nodes-base.errorTrigger' || 
      node.name.toLowerCase().includes('error')
    );
    
    if (!hasErrorHandler) {
      errors.push(`${filePath}: Workflow sin manejo de errores`);
    }
    
    return errors;
  }

  async pathExists(path) {
    try {
      await fs.access(path);
      return true;
    } catch {
      return false;
    }
  }
}

new WorkflowValidator().validateAll();
```

```bash
- Bug fixes in scheduling system"

# 5. Merge a staging para testing
git checkout staging

git push origin staging

# 6. Tras validación en staging, merge a main
git checkout main
git merge release/v2.1.0
git tag -a v2.1.0 -m "Release v2.1.0"
git push origin main --tags

# 7. Cleanup
git branch -d release/v2.1.0
git push origin --delete release/v2.1.0
```

```javascript
// scripts/monitor-workflows.js
const axios = require('axios');

class WorkflowMonitor {
  async checkAllEnvironments() {
    const environments = ['development', 'staging', 'production'];
    
    for (const env of environments) {
      try {
        await this.checkEnvironmentHealth(env);
      } catch (error) {
        await this.createGitLabIssue(env, error);
      }
    }
  }

  async checkEnvironmentHealth(env) {
    const baseURL = process.env[`N8N_${env.toUpperCase()}_URL`];
    const apiKey = process.env[`N8N_${env.toUpperCase()}_API_KEY`];
    
    // Check API health
    const healthResponse = await axios.get(`${baseURL}/healthz`);
    if (healthResponse.status !== 200) {
      throw new Error(`${env} API health check failed`);
    }
    
    // Check workflows status
    const workflowsResponse = await axios.get(`${baseURL}/api/v1/workflows`, {
      headers: { 'X-N8N-API-KEY': apiKey }
    });
    
    const activeWorkflows = workflowsResponse.data.data.filter(w => w.active);
    const failedWorkflows = await this.checkWorkflowExecutions(baseURL, apiKey, activeWorkflows);
    
    if (failedWorkflows.length > 0) {
      throw new Error(`${failedWorkflows.length} workflows failing in ${env}`);
    }
  }

  async createGitLabIssue(env, error) {
    const issue = {
      title: `🚨 Workflow Issue Detected - ${env}`,
      description: `
**Environment**: ${env}
**Error**: ${error.message}
**Detection Time**: ${new Date().toISOString()}
**Pipeline**: ${process.env.CI_PIPELINE_URL || 'Manual execution'}

This issue was automatically created by the monitoring system.

/label ~"type::bug" ~"priority::high" ~"environment::${env}"
/assign @on-call-team
      `,
      labels: ['type::bug', 'priority::high', `environment::${env}`]
    };
    
    // Crear issue en GitLab via API
    console.log(`🚨 Creating GitLab issue for ${env}: ${error.message}`);
  }
}

// Ejecutar monitorización
new WorkflowMonitor().checkAllEnvironments();
```
